create table if not exists public.media_items ( id uuid primary key default gen_random_uuid(), user_id uuid not null references auth.users(id) on delete cascade, group_name text not null, category_name text not null, media_type text not null check (media_type in ('photo', 'video')), storage_path text not null unique, original_name text, created_at timestamptz not null default now() ); alter table public.media_items enable row level security; grant select, insert, update, delete on public.media_items to authenticated; drop policy if exists "kpop_media_items_select_own" on public.media_items; create policy "kpop_media_items_select_own" on public.media_items for select to authenticated using ((select auth.uid()) = user_id); drop policy if exists "kpop_media_items_insert_own" on public.media_items; create policy "kpop_media_items_insert_own" on public.media_items for insert to authenticated with check ((select auth.uid()) = user_id); drop policy if exists "kpop_media_items_update_own" on public.media_items; create policy "kpop_media_items_update_own" on public.media_items for update to authenticated using ((select auth.uid()) = user_id) with check ((select auth.uid()) = user_id); drop policy if exists "kpop_media_items_delete_own" on public.media_items; create policy "kpop_media_items_delete_own" on public.media_items for delete to authenticated using ((select auth.uid()) = user_id); insert into storage.buckets (id, name, public, file_size_limit) values ('kpop-media', 'kpop-media', false, 52428800) on conflict (id) do update set public = false; drop policy if exists "kpop_media_files_select_own" on storage.objects; create policy "kpop_media_files_select_own" on storage.objects for select to authenticated using ( bucket_id = 'kpop-media' and (storage.foldername(name))[1] = (select auth.uid()::text) ); drop policy if exists "kpop_media_files_insert_own" on storage.objects; create policy "kpop_media_files_insert_own" on storage.objects for insert to authenticated with check ( bucket_id = 'kpop-media' and (storage.foldername(name))[1] = (select auth.uid()::text) ); drop policy if exists "kpop_media_files_delete_own" on storage.objects; create policy "kpop_media_files_delete_own" on storage.objects for delete to authenticated using ( bucket_id = 'kpop-media' and (storage.foldername(name))[1] = (select auth.uid()::text) );
